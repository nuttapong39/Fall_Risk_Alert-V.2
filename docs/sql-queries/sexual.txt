-- MedAlert · Query ดึงผู้ป่วย: ความรุนแรงทางเพศ/STI (lab 2811)
-- จาก HOSxP (PostgreSQL / XE4) — รันใน Navicat for PostgreSQL
-- ใช้ตรวจสอบว่าระบบจะดึงเคสไหนเข้าคิวแจ้งเตือน module 'sexual'
--
-- ค่าที่ปรับได้:
--   • ช่วงวันที่  BETWEEN '2026-08-01' AND '2026-08-31'  (ตอนนี้ = เดือนปัจจุบัน)
--   • รหัส lab/pttype/icode  ← แก้ให้ตรงรหัสใน HOSxP ของ รพ. ท่าน
-- (query นี้ดึงจากตัวจริงของระบบอัตโนมัติ — ตรงกับที่ worker ใช้)
-- ============================================================

SELECT
              h.vn,
              h.hn,
              CONCAT(COALESCE(pt.pname,''), COALESCE(pt.fname,''), ' ', COALESCE(pt.lname,'')) AS fullname,
              pt.cid,
              pt.hometel,
              pt.addrpart AS address,
              EXTRACT(YEAR FROM age(h.order_date, pt.birthday))  AS age,
              CASE WHEN pt.sex='1' THEN 'ชาย' WHEN pt.sex='2' THEN 'หญิง' ELSE '' END AS sex,
              h.order_date  AS lab_date,
              h.report_time AS lab_time,
              l.lab_items_name_ref,
              l.lab_order_result,
              l.lab_order_number
            FROM   lab_order   l
            INNER JOIN lab_head  h  ON h.lab_order_number = l.lab_order_number
            LEFT  JOIN patient   pt ON pt.hn              = h.hn
            WHERE  l.lab_items_code = '2811'
            AND    h.order_date     BETWEEN '2026-08-01' AND '2026-08-31'
            AND    l.lab_order_result IS NOT NULL
            AND    l.lab_order_result <> ''
            ORDER  BY h.order_date DESC
